AdaptiveWardrobeManager::AdaptiveWardrobeManager() {
    std::random_device rd;
    randomEngine = std::mt19937(rd());
}

AdaptiveWardrobeManager* AdaptiveWardrobeManager::GetSingleton() {
    static AdaptiveWardrobeManager instance;
    return &instance;
}

void AdaptiveWardrobeManager::LoadWardrobe(const std::vector<AdaptiveOutfit>& outfits) {
    std::lock_guard<std::mutex> lock(wardrobeMutex);
    wardrobe = outfits;
}

std::vector<AdaptiveOutfit> AdaptiveWardrobeManager::GetWardrobe() {
    std::lock_guard<std::mutex> lock(wardrobeMutex);
    return wardrobe;
}

ContextFlags AdaptiveWardrobeManager::ResolveContextHierarchy(ContextFlags activeContext) const {
    if (HasContext(activeContext, ContextFlags::Combat)) return ContextFlags::Combat;
    if (HasContext(activeContext, ContextFlags::Swim)) return ContextFlags::Swim;
    if (HasContext(activeContext, ContextFlags::PlayerHome)) return ContextFlags::PlayerHome;
    if (HasContext(activeContext, ContextFlags::CityTown)) return ContextFlags::CityTown;
    return ContextFlags::None;
}

float AdaptiveWardrobeManager::CalculatePreferenceWeight(uint8_t stars) const {
    static constexpr float kWeightTable[] = { 4.0f, 1.0f, 2.0f, 4.0f, 7.0f, 11.0f };
    if (stars <= 5) return kWeightTable[stars];
    return 4.0f;
}

float AdaptiveWardrobeManager::CalculateRepetitionPenalty(const std::string& id) const {
    size_t count = 0;
    for (const auto& historyID : recentHistory) {
        if (historyID == id) {
            count++;
        }
    }
    if (count == 0) return 1.00f;
    if (count == 1) return 0.70f;
    if (count == 2) return 0.40f;
    if (count == 3) return 0.20f;
    return 0.10f;
}

std::optional<SelectionResult> AdaptiveWardrobeManager::SelectOptimalOutfit(ContextFlags activeContext, [[maybe_unused]] bool isEmergency) {
    std::lock_guard<std::mutex> lock(wardrobeMutex);
    if (wardrobe.empty()) return std::nullopt;

    ContextFlags targetContext = ResolveContextHierarchy(activeContext);
    if (targetContext == ContextFlags::None) return std::nullopt;

    if (!currentOutfitID.empty()) {
        for (const auto& outfit : wardrobe) {
            if (outfit.id == currentOutfitID && outfit.controlState != ControlState::Excluded) {
                for (const auto& rule : outfit.suitabilities) {
                    if (HasContext(rule.contextMask, targetContext)) {
                        SelectionResult res;
                        res.selectedOutfit = outfit;
                        res.reasonCode = "Context Stability Maintained";
                        res.calculatedWeight = 100.0f;
                        return res;
                    }
                }
            }
        }
    }

    std::vector<const AdaptiveOutfit*> candidates;
    candidates.reserve(wardrobe.size());

    for (const auto& outfit : wardrobe) {
        if (outfit.controlState == ControlState::Excluded) continue;
        for (const auto& rule : outfit.suitabilities) {
            if (HasContext(rule.contextMask, targetContext)) {
                candidates.push_back(&outfit);
                break;
            }
        }
    }

    if (candidates.empty()) return std::nullopt;

    for (const auto* candidate : candidates) {
        if (candidate->controlState == ControlState::Forced) {
            SelectionResult res;
            res.selectedOutfit = *candidate;
            res.reasonCode = "Forced Control State Priority";
            res.calculatedWeight = 99999.0f;
            return res;
        }
    }

    std::vector<float> weights;
    weights.reserve(candidates.size());
    float totalWeight = 0.0f;

    for (const auto* candidate : candidates) {
        float prefWeight = CalculatePreferenceWeight(candidate->preferenceStars);
        float repPenalty = CalculateRepetitionPenalty(candidate->id);
        float ctrlMultiplier = (candidate->controlState == ControlState::Favorite) ? 1.5f : 1.0f;

        float weight = prefWeight * repPenalty * ctrlMultiplier;
        weights.push_back(weight);
        totalWeight += weight;

        SKSE::log::info("SelectOptimalOutfit: Candidate '{}' ({}) | Stars: {} | RepPenalty: {:.2f} | FinalWeight: {:.2f}",
                        candidate->name, candidate->id, candidate->preferenceStars, repPenalty, weight);
    }

    if (totalWeight <= 0.0f) return std::nullopt;

    std::uniform_real_distribution<float> dist(0.0f, totalWeight);
    float roll = dist(randomEngine);
    float currentSum = 0.0f;
    size_t selectedIndex = 0;

    for (size_t i = 0; i < candidates.size(); ++i) {
        currentSum += weights[i];
        if (roll <= currentSum) {
            selectedIndex = i;
            break;
        }
    }

    SelectionResult result;
    result.selectedOutfit = *candidates[selectedIndex];
    result.reasonCode = "Weighted Random Selection";
    result.calculatedWeight = weights[selectedIndex];

    SKSE::log::info("SelectOptimalOutfit: WINNER Selected '{}' ({}) | Weight: {:.2f} / Total: {:.2f} (Prob: {:.1f}%)",
                    result.selectedOutfit.name, result.selectedOutfit.id, result.calculatedWeight, totalWeight,
                    (result.calculatedWeight / totalWeight) * 100.0f);

    return result;
}

void AdaptiveWardrobeManager::RecordOutfitEquipped(const std::string& outfitID, uint32_t) {
    std::lock_guard<std::mutex> lock(wardrobeMutex);
    currentOutfitID = outfitID;

    recentHistory.push_front(outfitID);
    if (recentHistory.size() > 5) {
        recentHistory.pop_back();
    }
}

void AdaptiveWardrobeManager::PauseAutoSystem(uint32_t durationSeconds) {
    std::lock_guard<std::mutex> lock(wardrobeMutex);
    manualPauseUntil = std::chrono::steady_clock::now() + std::chrono::seconds(durationSeconds);
}

void AdaptiveWardrobeManager::ClearPause() {
    std::lock_guard<std::mutex> lock(wardrobeMutex);
    manualPauseUntil = std::chrono::steady_clock::time_point{};
}

bool AdaptiveWardrobeManager::IsPaused() const {
    std::lock_guard<std::mutex> lock(wardrobeMutex);
    return std::chrono::steady_clock::now() < manualPauseUntil;
}

void AdaptiveWardrobeManager::SetCurrentOutfitID(const std::string& id) {
    std::lock_guard<std::mutex> lock(wardrobeMutex);
    currentOutfitID = id;
}

    std::lock_guard<std::mutex> lock(wardrobeMutex);

std::vector<InventoryWornEntry> GetEquippedInventoryEntries() {
    std::vector<InventoryWornEntry> equipped;
    auto player = RE::PlayerCharacter::GetSingleton();
    if (!player) return equipped;

    auto inventory = player->GetInventory();
    for (const auto& [item, data] : inventory) {
        if (item && item->IsArmor() && data.second && data.second->extraLists) {
            auto* armor = item->As<RE::TESObjectARMO>();
            if (armor && armor->HasPartOf(RE::BIPED_MODEL::BipedObjectSlot::kShield)) {
                continue;
            }

            for (auto* extraList : *data.second->extraLists) {
                if (extraList && (extraList->HasType(RE::ExtraDataType::kWorn) || extraList->HasType(RE::ExtraDataType::kWornLeft))) {
                    InventoryWornEntry entry;
                    entry.object = item;
                    entry.extraList = extraList;
                    equipped.push_back(entry);
                }
            }
        }
    }
    return equipped;
}

void PerformOutfitSwap(RE::PlayerCharacter* player, RE::ActorEquipManager* equipManager, const std::vector<RE::FormID>& targetItemIDs) {
    if (!player || !equipManager) return;

    std::set<RE::FormID> targetSet(targetItemIDs.begin(), targetItemIDs.end());
    auto currentlyWornEntries = GetEquippedInventoryEntries();
    std::set<RE::FormID> currentlyWornSet;
    for (const auto& entry : currentlyWornEntries) {
        if (entry.object) {
            currentlyWornSet.insert(entry.object->GetFormID());
        }
    }

    for (const auto& entry : currentlyWornEntries) {
        if (entry.object && entry.object->IsArmor()) {
            RE::FormID formID = entry.object->GetFormID();
            if (targetSet.find(formID) == targetSet.end()) {
                equipManager->UnequipObject(player, entry.object, entry.extraList, 1, nullptr, false, false, false, true);
            }
        }
    }

    auto freshInventory = player->GetInventory();
    for (RE::FormID formID : targetItemIDs) {
        if (currentlyWornSet.find(formID) == currentlyWornSet.end()) {
            auto* form = RE::TESForm::LookupByID(formID);
            auto* boundObj = form ? form->As<RE::TESBoundObject>() : nullptr;
            if (!boundObj || !boundObj->IsArmor()) {
                continue;
            }
            auto it = freshInventory.find(boundObj);
            bool playerOwnsItem = (it != freshInventory.end()) && (it->second.first > 0);
            if (!playerOwnsItem) {
                continue;
            }

            RE::ExtraDataList* targetExtra = nullptr;
            if (it->second.second && it->second.second->extraLists && !it->second.second->extraLists->empty()) {
                targetExtra = it->second.second->extraLists->front();
            }
            equipManager->EquipObject(player, boundObj, targetExtra, 1, nullptr, false, false, false, true);
        }
    }
}

void SaveOutfitToSlotImpl(size_t wheelIdx, size_t slotIdx) {
    auto equipped = GetEquippedInventoryEntries();
    if (equipped.empty()) {
        RE::DebugNotification("No equipped armor/clothing to save!");
        return;
    }

    std::lock_guard<std::mutex> lock(g_dataMutex);
    if (wheelIdx < g_manager.wheels.size() && slotIdx < g_manager.wheels[wheelIdx].slots.size()) {
        Outfit outfit;
        if (g_manager.wheels[wheelIdx].slots[slotIdx].outfit.has_value()) {
            outfit.name = g_manager.wheels[wheelIdx].slots[slotIdx].outfit->name;
            outfit.contextTag = g_manager.wheels[wheelIdx].slots[slotIdx].outfit->contextTag;
            outfit.preferenceStars = g_manager.wheels[wheelIdx].slots[slotIdx].outfit->preferenceStars;
            outfit.controlState = g_manager.wheels[wheelIdx].slots[slotIdx].outfit->controlState;
            outfit.usageCount = g_manager.wheels[wheelIdx].slots[slotIdx].outfit->usageCount;
            outfit.lastWornTimestamp = g_manager.wheels[wheelIdx].slots[slotIdx].outfit->lastWornTimestamp;
        } else {
            outfit.name = "Outfit " + std::to_string(slotIdx + 1);
            outfit.contextTag = "";
            outfit.preferenceStars = 3;
            outfit.controlState = ControlState::Normal;
            outfit.usageCount = 0;
            outfit.lastWornTimestamp = 0;
        }
        for (const auto& entry : equipped) {
            if (entry.object && entry.object->IsArmor()) {
                outfit.itemIDs.push_back(entry.object->GetFormID());
            }
        }
        g_manager.wheels[wheelIdx].slots[slotIdx].outfit = outfit;
        SaveDataInternal();
        RE::DebugNotification(("Saved " + outfit.name).c_str());
    }
}

void EquipSlotImpl(size_t wheelIdx, size_t slotIdx) {
    Outfit targetOutfit;
    bool found = false;
    {
        std::lock_guard<std::mutex> lock(g_dataMutex);
        if (wheelIdx < g_manager.wheels.size() && slotIdx < g_manager.wheels[wheelIdx].slots.size()) {
            if (g_manager.wheels[wheelIdx].slots[slotIdx].outfit.has_value()) {
                targetOutfit = *g_manager.wheels[wheelIdx].slots[slotIdx].outfit;
                found = true;
            }
        }
    }

    if (!found) return;

    auto player = RE::PlayerCharacter::GetSingleton();
    auto equipManager = RE::ActorEquipManager::GetSingleton();
    if (!player || !equipManager) return;

    PerformOutfitSwap(player, equipManager, targetOutfit.itemIDs);

    std::string adaptiveOutfitID = std::to_string(wheelIdx) + "_" + std::to_string(slotIdx);
    AdaptiveWardrobeManager::GetSingleton()->RecordOutfitEquipped(adaptiveOutfitID, 0);
    AdaptiveWardrobeManager::GetSingleton()->PauseAutoSystem(10);

    RE::DebugNotification(("Equipped " + targetOutfit.name).c_str());

void ExecutionQueue::RegisterEventsInternal() {
    if (g_eventsRegistered.load(std::memory_order_acquire)) return;

    auto scriptSource = RE::ScriptEventSourceHolder::GetSingleton();
    auto player = RE::PlayerCharacter::GetSingleton();
    auto ui = RE::UI::GetSingleton();

    if (!scriptSource || !player || !ui) return;

    scriptSource->AddEventSink<RE::TESCombatEvent>(this);
    scriptSource->AddEventSink<RE::TESActorLocationChangeEvent>(this);
    scriptSource->AddEventSink<RE::TESCellFullyLoadedEvent>(this);
    ui->AddEventSink<RE::MenuOpenCloseEvent>(this);

    auto* cellEventSource = player->AsBGSActorCellEventSource();
    if (cellEventSource) {
        cellEventSource->AddEventSink(this);
    }

    g_eventsRegistered.store(true, std::memory_order_release);
    SKSE::log::info("RegisterEvents: SUCCESS");
}

void ExecutionQueue::RegisterEvents() {
    SKSE::GetTaskInterface()->AddTask([this]() {
        RegisterEventsInternal();
    });
}

ContextFlags ExecutionQueue::ResolveLocationContext(RE::BGSLocation* location, RE::TESObjectCELL* cell) {
    if (location) {
        if (location->HasKeywordString("LocTypePlayerHouse") || (location->parentLoc && location->parentLoc->HasKeywordString("LocTypePlayerHouse"))) {
            return ContextFlags::PlayerHome;
        }
        if (location->HasKeywordString("LocTypeCity") || location->HasKeywordString("LocTypeTown") ||
            (location->parentLoc && (location->parentLoc->HasKeywordString("LocTypeCity") || location->parentLoc->HasKeywordString("LocTypeTown")))) {
            return ContextFlags::CityTown;
        }
    }

    if (cell && cell->IsInteriorCell()) {
        auto owner = cell->GetOwner();
        if (owner) {
            if (owner->Is(RE::FormType::NPC) && owner->GetFormID() == 0x00000007) {
                return ContextFlags::PlayerHome;
            }
            if (owner->Is(RE::FormType::Faction)) {
                uint32_t fID = owner->GetFormID();
                if (fID == 0x00013702 || fID == 0x000A2C9D) {
                    return ContextFlags::PlayerHome;
                }
            }
        }
    }

    return ContextFlags::None;
}

ContextFlags ExecutionQueue::GatherCurrentContext() {
    auto player = RE::PlayerCharacter::GetSingleton();
    if (!player) return ContextFlags::None;

    ContextFlags flags = ContextFlags::None;
    auto parentCell = player->GetParentCell();
    auto loc = player->GetCurrentLocation();

    flags |= ResolveLocationContext(loc, parentCell);

    if (player->IsInCombat()) {
        flags |= ContextFlags::Combat;
    }

    auto actorState = player->AsActorState();
    float waterHeight = player->GetWaterHeight();
    float playerZ = player->GetPositionZ();
    if ((actorState && actorState->IsSwimming()) || (waterHeight > playerZ + 20.0f)) {
        flags |= ContextFlags::Swim;
    }

    return flags;
}

RE::BSEventNotifyControl ExecutionQueue::ProcessEvent(const RE::TESCombatEvent* a_event, RE::BSTEventSource<RE::TESCombatEvent>*) {
    if (!a_event || !a_event->targetActor || !a_event->targetActor->IsPlayerRef()) {
        return RE::BSEventNotifyControl::kContinue;
    }
    SKSE::log::info("ProcessEvent: TESCombatEvent fired");
    AdaptiveWardrobeManager::GetSingleton()->ClearPause();
    EvaluateAndQueueChange(a_event->newState.get() == RE::ACTOR_COMBAT_STATE::kCombat);
    return RE::BSEventNotifyControl::kContinue;
}

RE::BSEventNotifyControl ExecutionQueue::ProcessEvent(const RE::BGSActorCellEvent* a_event, RE::BSTEventSource<RE::BGSActorCellEvent>*) {
    if (!a_event) return RE::BSEventNotifyControl::kContinue;
    SKSE::log::info("ProcessEvent: BGSActorCellEvent fired");
    AdaptiveWardrobeManager::GetSingleton()->ClearPause();
    EvaluateAndQueueChange(false);
    return RE::BSEventNotifyControl::kContinue;
}

RE::BSEventNotifyControl ExecutionQueue::ProcessEvent(const RE::TESActorLocationChangeEvent* a_event, RE::BSTEventSource<RE::TESActorLocationChangeEvent>*) {
    if (!a_event || !a_event->actor || !a_event->actor->IsPlayerRef()) {
        return RE::BSEventNotifyControl::kContinue;
    }
    SKSE::log::info("ProcessEvent: TESActorLocationChangeEvent fired");
    AdaptiveWardrobeManager::GetSingleton()->ClearPause();
    EvaluateAndQueueChange(false);
    return RE::BSEventNotifyControl::kContinue;
}

RE::BSEventNotifyControl ExecutionQueue::ProcessEvent(const RE::TESCellFullyLoadedEvent* a_event, RE::BSTEventSource<RE::TESCellFullyLoadedEvent>*) {
    if (!a_event) return RE::BSEventNotifyControl::kContinue;
    SKSE::log::info("ProcessEvent: TESCellFullyLoadedEvent fired - cell fully ready");
    AdaptiveWardrobeManager::GetSingleton()->ClearPause();
    EvaluateAndQueueChange(false);
    return RE::BSEventNotifyControl::kContinue;
}

RE::BSEventNotifyControl ExecutionQueue::ProcessEvent(const RE::MenuOpenCloseEvent* a_event, RE::BSTEventSource<RE::MenuOpenCloseEvent>*) {
    if (!a_event) return RE::BSEventNotifyControl::kContinue;

    if (!a_event->opening && (a_event->menuName == RE::LoadingMenu::MENU_NAME || a_event->menuName == RE::FaderMenu::MENU_NAME)) {
        SKSE::log::info("ProcessEvent: MenuOpenCloseEvent - '{}' CLOSED", a_event->menuName.c_str());
        AdaptiveWardrobeManager::GetSingleton()->ClearPause();
        EvaluateAndQueueChange(false);
    }
    return RE::BSEventNotifyControl::kContinue;
}

void ExecutionQueue::EvaluateAndQueueChange(bool isEmergency) {
    if (g_skipInitialLoadEval.exchange(false, std::memory_order_acq_rel)) {
        SKSE::log::info("EvaluateAndQueueChange: Skipped - Initial game load");
        return;
    }

    uint32_t currentTick = GetTickCount();
    uint32_t lastTick = g_lastEvalTick.load(std::memory_order_acquire);
    if (!isEmergency && (currentTick - lastTick < 1500)) {
        SKSE::log::info("EvaluateAndQueueChange: Coalesced duplicate cell/location event");
        return;
    }
    g_lastEvalTick.store(currentTick, std::memory_order_release);

    if (!g_gameFullyLoaded.load(std::memory_order_acquire)) {
        SKSE::log::info("EvaluateAndQueueChange: Skipped - Game not fully loaded");
        return;
    }

    auto uiRef = RE::UI::GetSingleton();
    if (uiRef && (uiRef->IsMenuOpen(RE::MainMenu::MENU_NAME) || uiRef->IsMenuOpen(RE::LoadingMenu::MENU_NAME) || uiRef->IsMenuOpen(RE::FaderMenu::MENU_NAME))) {
        SKSE::log::info("EvaluateAndQueueChange: Skipped - Loading/Fader/Main menu active");
        return;
    }

    auto playerRef = RE::PlayerCharacter::GetSingleton();
    if (!playerRef || !playerRef->Is3DLoaded()) {
        SKSE::log::info("EvaluateAndQueueChange: Skipped - Player or 3D not loaded");
        return;
    }

    auto manager = AdaptiveWardrobeManager::GetSingleton();
    if (manager->IsPaused()) {
        SKSE::log::info("EvaluateAndQueueChange: Skipped - System is paused");
        return;
    }

    ContextFlags currentContext = GatherCurrentContext();
    SKSE::log::info("EvaluateAndQueueChange: Context resolved to {:08X}", static_cast<uint32_t>(currentContext));

    auto selectionOpt = manager->SelectOptimalOutfit(currentContext, isEmergency);
    if (!selectionOpt.has_value()) {
        SKSE::log::info("EvaluateAndQueueChange: Skipped - No optimal outfit found for context {:08X}", static_cast<uint32_t>(currentContext));
        return;
    }

    auto selection = selectionOpt.value();
    SKSE::log::info("EvaluateAndQueueChange: Candidate selected outfit ID '{}' ('{}'), current outfit ID '{}'",
                    selection.selectedOutfit.id, selection.selectedOutfit.name, manager->GetCurrentOutfitID());

    if (selection.selectedOutfit.id == manager->GetCurrentOutfitID()) {
        SKSE::log::info("EvaluateAndQueueChange: Skipped - Selected outfit matches current outfit ID '{}'", manager->GetCurrentOutfitID());
        return;
    }

    std::string targetID = selection.selectedOutfit.id;
    std::string targetName = selection.selectedOutfit.name;
    std::vector<RE::FormID> targetItems = selection.selectedOutfit.itemIDs;

    SKSE::log::info("EvaluateAndQueueChange: Queuing SKSE task to swap to '{}' ({})", targetName, targetID);

    SKSE::GetTaskInterface()->AddTask([manager, targetID, targetName, targetItems]() {
        auto player = RE::PlayerCharacter::GetSingleton();
        auto equipManager = RE::ActorEquipManager::GetSingleton();
        if (!player || !equipManager || !player->Is3DLoaded()) {
            SKSE::log::info("TaskLambda: Skipped - Player/EquipManager null or 3D not loaded");
            return;
        }

        if (manager->GetCurrentOutfitID() == targetID) {
            SKSE::log::info("TaskLambda: Skipped - Outfit ID '{}' already equipped", targetID);
            return;
        }

        SKSE::log::info("TaskLambda: EXECUTING PerformOutfitSwap to '{}' ({})", targetName, targetID);
        PerformOutfitSwap(player, equipManager, targetItems);
        manager->RecordOutfitEquipped(targetID, 0);
        SKSE::log::info("TaskLambda: PerformOutfitSwap COMPLETE for '{}'", targetName);
    });

static inline bool IsModifierMatch(uint32_t modKey, bool shift, bool ctrl, bool alt) {
    if (modKey == 0) return true;
    if ((modKey == 0x2A || modKey == 0x36 || modKey == 42) && shift) return true;
    if ((modKey == 0x1D || modKey == 0x9D || modKey == 29) && ctrl) return true;
    if ((modKey == 0x38 || modKey == 0xB8 || modKey == 56) && alt) return true;
    return false;
}

void ShowWheel(bool isEditMode) {
    if (!PrismaUI || view == 0) {
        SKSE::log::error("ShowWheel failed: PrismaUI API or View is null");
        return;
    }

    std::string wheelDataJson = GetWheelJson(isEditMode);

    SKSE::log::info("ShowWheel: Showing view");
    PrismaUI->Show(view);

    SKSE::log::info("ShowWheel: Focusing view");
    PrismaUI->Focus(view, true, false);

    SKSE::log::info("ShowWheel: Calling InteropCall openWheel");
    PrismaUI->InteropCall(view, "openWheel", wheelDataJson.c_str());

    isMenuOpen.store(true, std::memory_order_release);
}

void HideWheel() {
    if (PrismaUI && view != 0) {
        PrismaUI->Hide(view);
    }
    isMenuOpen.store(false, std::memory_order_release);
}

RE::BSEventNotifyControl WheelInputHandler::ProcessEvent(RE::InputEvent* const* a_event, RE::BSTEventSource<RE::InputEvent*>*) {
    if (!a_event || !*a_event) return RE::BSEventNotifyControl::kContinue;

    auto player = RE::PlayerCharacter::GetSingleton();
    if (!player || !player->Is3DLoaded()) {
        return RE::BSEventNotifyControl::kContinue;
    }

    auto ui = RE::UI::GetSingleton();
    if (!ui) return RE::BSEventNotifyControl::kContinue;

    bool currentlyOpen = isMenuOpen.load(std::memory_order_acquire);

    for (auto event = *a_event; event; event = event->next) {
        if (event->eventType == RE::INPUT_EVENT_TYPE::kButton) {
            auto buttonEvent = event->AsButtonEvent();
            if (buttonEvent && buttonEvent->IsPressed()) {
                auto device = buttonEvent->GetDevice();
                uint32_t keyCode = buttonEvent->GetIDCode();

                if (device == RE::INPUT_DEVICE::kKeyboard && keyCode == 0x01 && currentlyOpen) {
                    SKSE::GetTaskInterface()->AddTask([]() {
                        HideWheel();
                    });
                    return RE::BSEventNotifyControl::kContinue;
                }
            }
        }
    }

    if (g_isTyping.load(std::memory_order_acquire)) {
        return RE::BSEventNotifyControl::kContinue;
    }

    bool isInventoryOpen = ui->IsMenuOpen(MENU_INVENTORY);

    bool hasBlockingMenu = ui->IsMenuOpen(MENU_CONSOLE) ||
                          ui->IsMenuOpen(MENU_RACEMENU) ||
                          ui->IsMenuOpen(MENU_MAGIC) ||
                          ui->IsMenuOpen(MENU_CONTAINER) ||
                          ui->IsMenuOpen(MENU_BARTER) ||
                          ui->IsMenuOpen(MENU_BOOK) ||
                          ui->IsMenuOpen(MENU_JOURNAL) ||
                          ui->IsMenuOpen(MENU_MAP) ||
                          ui->IsMenuOpen(MENU_LOCKPICKING) ||
                          ui->IsMenuOpen(MENU_CRAFTING) ||
                          ui->IsMenuOpen(MENU_DIALOGUE) ||
                          ui->IsMenuOpen(MENU_FADER) ||
                          ui->IsApplicationMenuOpen() ||
                          ui->IsModalMenuOpen();

    if (hasBlockingMenu) {
        return RE::BSEventNotifyControl::kContinue;
    }

    if (ui->GameIsPaused() && !currentlyOpen && !isInventoryOpen) {
        return RE::BSEventNotifyControl::kContinue;
    }

    bool shift = (GetKeyState(VK_SHIFT) & 0x8000) != 0 || (GetKeyState(VK_LSHIFT) & 0x8000) != 0 || (GetKeyState(VK_RSHIFT) & 0x8000) != 0;
    bool ctrl = (GetKeyState(VK_CONTROL) & 0x8000) != 0 || (GetKeyState(VK_LCONTROL) & 0x8000) != 0 || (GetKeyState(VK_RCONTROL) & 0x8000) != 0;
    bool alt = (GetKeyState(VK_MENU) & 0x8000) != 0 || (GetKeyState(VK_LMENU) & 0x8000) != 0 || (GetKeyState(VK_RMENU) & 0x8000) != 0;

    for (auto event = *a_event; event; event = event->next) {
        if (event->eventType == RE::INPUT_EVENT_TYPE::kButton) {
            auto buttonEvent = event->AsButtonEvent();
            if (buttonEvent && buttonEvent->IsPressed()) {
                auto device = buttonEvent->GetDevice();
                if (device != RE::INPUT_DEVICE::kKeyboard) {
                    continue;
                }

                uint32_t keyCode = buttonEvent->GetIDCode();

                bool editRequiresMod = (g_editModifierKey != 0);
                bool playRequiresMod = (g_playModifierKey != 0);

                bool editModMatch = IsModifierMatch(g_editModifierKey, shift, ctrl, alt);
                bool playModMatch = IsModifierMatch(g_playModifierKey, shift, ctrl, alt);

                bool triggerEdit = false;
                bool triggerPlay = false;

                if (keyCode == g_editKey && editModMatch && editRequiresMod) {
                    triggerEdit = true;
                } else if (keyCode == g_playKey && playModMatch && playRequiresMod) {
                    triggerPlay = true;
                } else if (keyCode == g_editKey && editModMatch && !editRequiresMod) {
                    triggerEdit = true;
                } else if (keyCode == g_playKey && playModMatch && !playRequiresMod) {
                    triggerPlay = true;
                }

                if (triggerEdit || triggerPlay) {
                    auto now = std::chrono::steady_clock::now();
                    auto elapsedMs = std::chrono::duration_cast<std::chrono::milliseconds>(now - g_lastToggleTime).count();

                    if (elapsedMs < 200) {
                        return RE::BSEventNotifyControl::kContinue;
                    }

                    g_lastToggleTime = now;
                    bool targetMode = triggerEdit;

                    SKSE::GetTaskInterface()->AddTask([targetMode]() {
                        if (!isMenuOpen.load(std::memory_order_acquire)) {
                            ShowWheel(targetMode);
                        } else {
                            HideWheel();
                        }
                    });
                }
            }
        }
    }
    return RE::BSEventNotifyControl::kContinue;
}#include "JSInterop.h"

void ChangeWheelListener(const char* data) {
    SKSE::log::info("ChangeWheel listener triggered with payload: {}", data ? data : "null");
    try {
        json req = json::parse(data);
        int delta = req.value("delta", 0);
        bool editMode = req.value("editMode", false);

        SKSE::GetTaskInterface()->AddTask([delta, editMode]() {
            std::lock_guard<std::mutex> lock(g_dataMutex);
            if (g_manager.wheels.empty()) return;
            int current = static_cast<int>(g_manager.activeWheelIndex);
            int total = static_cast<int>(g_manager.wheels.size());
            current = (current + delta) % total;
            if (current < 0) current += total;
            g_manager.activeWheelIndex = static_cast<size_t>(current);
            SaveDataInternal();

            std::string jsCmd = GetWheelJsonInternal(editMode);
            PrismaUI->InteropCall(view, "openWheel", jsCmd.c_str());
        });
    } catch (const std::exception& e) {
        SKSE::log::error("Exception in ChangeWheel listener: {}", e.what());

static void SKSEMessageHandler(SKSE::MessagingInterface::Message* message) {
    if (!message) return;
    SKSE::log::info("Received SKSE message type: {}", message->type);

    if (message->type == SKSE::MessagingInterface::kDataLoaded) {
        SKSE::log::info("Processing kDataLoaded event");
        LoadINI();
        LoadData();

        if (!PrismaUI) {
            SKSE::log::info("Requesting PrismaUI API Interface v2 at kDataLoaded");
            PrismaUI = PRISMA_UI_API::RequestPluginAPI<PRISMA_UI_API::IVPrismaUI2>();
            if (!PrismaUI) {
                SKSE::log::warn("IVPrismaUI2 request failed, trying IVPrismaUI1");
                PrismaUI = static_cast<PRISMA_UI_API::IVPrismaUI2*>(PRISMA_UI_API::RequestPluginAPI(PRISMA_UI_API::InterfaceVersion::V1));
            }
            SKSE::log::info("PrismaUI API pointer: {}", static_cast<void*>(PrismaUI));
        }

        if (PrismaUI && view == 0) {
            const char* viewPath = "PrismaOutfitWheel/index.html";
            SKSE::log::info("Calling CreateView with path: '{}'", viewPath);

            view = PrismaUI->CreateView(viewPath, OnDomReadyCallbackFunc);
            SKSE::log::info("CreateView returned View ID: {}", view);

            PrismaUI->Hide(view);
            isMenuOpen.store(false, std::memory_order_release);

            PrismaUI->RegisterConsoleCallback(view, OnConsoleMessageCallbackFunc);
            PrismaUI->RegisterJSListener(view, "SaveSettings", SaveSettingsListener);
            PrismaUI->RegisterJSListener(view, "SetTypingMode", SetTypingModeListener);
            PrismaUI->RegisterJSListener(view, "SaveOutfitToSlot", SaveOutfitListener);
            PrismaUI->RegisterJSListener(view, "RenameSlot", RenameSlotListener);
            PrismaUI->RegisterJSListener(view, "RenameWheel", RenameWheelListener);
            PrismaUI->RegisterJSListener(view, "ClearSlot", ClearSlotListener);
            PrismaUI->RegisterJSListener(view, "EquipSlot", EquipSlotListener);
            PrismaUI->RegisterJSListener(view, "OnCloseWheel", OnCloseWheelListener);
            PrismaUI->RegisterJSListener(view, "ChangeWheel", ChangeWheelListener);
            PrismaUI->RegisterJSListener(view, "AddWheel", AddWheelListener);
            PrismaUI->RegisterJSListener(view, "AddSlot", AddSlotListener);
            PrismaUI->RegisterJSListener(view, "DeleteSlot", DeleteSlotListener);
            PrismaUI->RegisterJSListener(view, "DeleteWheel", DeleteWheelListener);
            PrismaUI->RegisterJSListener(view, "SetSlotContext", SetSlotContextListener);
            PrismaUI->RegisterJSListener(view, "SetSlotRating", SetSlotRatingListener);
            SKSE::log::info("Registered all JS listeners");
        }

        WheelInputHandler::Register();
    } else if (message->type == SKSE::MessagingInterface::kPreLoadGame) {
        SKSE::log::info("PreLoadGame message received - invalidating pending tasks and resetting fully loaded flag");
        g_gameFullyLoaded.store(false, std::memory_order_release);
        g_gameSessionGeneration.fetch_add(1, std::memory_order_acq_rel);
    } else if (message->type == SKSE::MessagingInterface::kPostLoadGame || message->type == SKSE::MessagingInterface::kNewGame) {
        SKSE::log::info("Player loaded into game world - enabling adaptive wardrobe auto-system");
        LoadData();
        g_gameFullyLoaded.store(true, std::memory_order_release);
        ExecutionQueue::GetSingleton()->RegisterEvents();
    }
}

extern "C" DLLEXPORT bool SKSEAPI SKSEPlugin_Load(const SKSE::LoadInterface* a_skse) {
    InitLog();
    SKSE::log::info("SKSEPlugin_Load called");

    REL::Module::reset();
    auto g_messaging = reinterpret_cast<SKSE::MessagingInterface*>(a_skse->QueryInterface(SKSE::LoadInterface::kMessaging));
    if (!g_messaging) {
        SKSE::log::error("Failed to query SKSE MessagingInterface");
        return false;
    }

    SKSE::Init(a_skse);
    SKSE::AllocTrampoline(1 << 10);
    g_messaging->RegisterListener("SKSE", SKSEMessageHandler);
    SKSE::log::info("SKSE message listener registered successfully");

    return true;

static std::string ToLowerString(std::string str) {
    std::transform(str.begin(), str.end(), str.begin(), [](unsigned char c) {
        return static_cast<char>(std::tolower(c));
    });
    return str;
}

static ContextFlags ParseContextTagToFlags(const std::string& tag) {
    std::string lower = ToLowerString(tag);
    ContextFlags flags = ContextFlags::None;

    if (lower.find("city") != std::string::npos || lower.find("town") != std::string::npos) {
        flags = flags | ContextFlags::CityTown;
    }
    if (lower.find("home") != std::string::npos || lower.find("house") != std::string::npos) {
        flags = flags | ContextFlags::PlayerHome;
    }
    if (lower.find("combat") != std::string::npos || lower.find("fight") != std::string::npos) {
        flags = flags | ContextFlags::Combat;
    }
    if (lower.find("swim") != std::string::npos || lower.find("water") != std::string::npos) {
        flags = flags | ContextFlags::Swim;
    }

    return flags;
}

void SyncWheelOutfitsToAdaptiveWardrobe() {
    std::vector<AdaptiveOutfit> adaptiveList;
    size_t wIdx = 0;
    for (const auto& wheel : g_manager.wheels) {
        size_t sIdx = 0;
        for (const auto& slot : wheel.slots) {
            if (slot.outfit.has_value()) {
                AdaptiveOutfit ao;
                ao.id = std::to_string(wIdx) + "_" + std::to_string(sIdx);
                ao.name = slot.outfit->name;
                ao.itemIDs = slot.outfit->itemIDs;
                ao.preferenceStars = slot.outfit->preferenceStars;
                ao.controlState = slot.outfit->controlState;
                ao.dynamicVars.usageCount = slot.outfit->usageCount;
                ao.dynamicVars.lastWornTimestamp = slot.outfit->lastWornTimestamp;

                ContextFlags cFlags = ParseContextTagToFlags(slot.outfit->contextTag);
                if (cFlags != ContextFlags::None) {
                    SuitabilityRule rule;
                    rule.contextMask = cFlags;
                    rule.score = static_cast<uint8_t>(3 + slot.outfit->preferenceStars);
                    ao.suitabilities.push_back(rule);
                }

                adaptiveList.push_back(ao);
            }
            sIdx++;
        }
        wIdx++;
    }
    AdaptiveWardrobeManager::GetSingleton()->LoadWardrobe(adaptiveList);
    SKSE::log::info("Synced {} wheel outfits into AdaptiveWardrobeManager", adaptiveList.size());
}

static json SerializeFormID(RE::FormID formID) {
    auto* form = RE::TESForm::LookupByID(formID);
    if (form) {
        auto* file = form->GetFile(0);
        if (file) {
            std::string modName = file->fileName;
            uint32_t localID = file->IsLight() ? (formID & 0x00000FFF) : (formID & 0x00FFFFFF);
            std::stringstream ss;
            ss << "0x" << std::uppercase << std::setfill('0') << std::setw(8) << std::hex << localID;
            return json{
                {"plugin", modName},
                {"localID", ss.str()},
                {"rawID", static_cast<uint32_t>(formID)}
            };
        }
    }
    std::stringstream ss;
    ss << "0x" << std::uppercase << std::setfill('0') << std::setw(8) << std::hex << formID;
    return json{
        {"plugin", "Skyrim.esm"},
        {"localID", ss.str()},
        {"rawID", static_cast<uint32_t>(formID)}
    };
}

static RE::FormID DeserializeFormID(const json& jItem) {
    if (jItem.is_number_integer() || jItem.is_number_unsigned()) {
        return jItem.get<RE::FormID>();
    }
    if (jItem.is_object()) {
        std::string plugin = jItem.value("plugin", "");
        uint32_t localID = 0;

        if (jItem.contains("localID")) {
            if (jItem.at("localID").is_number_unsigned() || jItem.at("localID").is_number_integer()) {
                localID = jItem.at("localID").get<uint32_t>();
            } else if (jItem.at("localID").is_string()) {
                std::string s = jItem.at("localID").get<std::string>();
                if (s.rfind("0x", 0) == 0 || s.rfind("0X", 0) == 0) {
                    s = s.substr(2);
                }
                try {
                    localID = static_cast<uint32_t>(std::stoul(s, nullptr, 16));
                } catch (...) {
                    localID = 0;
                }
            }
        }

        if (!plugin.empty() && localID != 0) {
            auto* dataHandler = RE::TESDataHandler::GetSingleton();
            if (dataHandler) {
                auto* boundObj = dataHandler->LookupForm<RE::TESBoundObject>(localID, plugin);
                if (boundObj) {
                    return boundObj->GetFormID();
                }
            }
        }

        if (jItem.contains("rawID")) {
            return jItem.value("rawID", static_cast<RE::FormID>(0));
        }
    }
    return 0;
}

void SaveDataInternal() {
    try {
        json root;
        root["activeWheel"] = g_manager.activeWheelIndex;
        json wheelsJson = json::array();
        for (const auto& wheel : g_manager.wheels) {
            json wJson;
            wJson["name"] = wheel.name;
            json slotsJson = json::array();
            for (const auto& slot : wheel.slots) {
                json sJson;
                if (slot.outfit.has_value()) {
                    sJson["hasOutfit"] = true;
                    sJson["name"] = slot.outfit->name;
                    sJson["contextTag"] = slot.outfit->contextTag;
                    sJson["preferenceStars"] = slot.outfit->preferenceStars;
                    sJson["controlState"] = static_cast<uint8_t>(slot.outfit->controlState);
                    sJson["usageCount"] = slot.outfit->usageCount;
                    sJson["lastWornTimestamp"] = slot.outfit->lastWornTimestamp;

                    json itemsArray = json::array();
                    for (auto fID : slot.outfit->itemIDs) {
                        itemsArray.push_back(SerializeFormID(fID));
                    }
                    sJson["items"] = itemsArray;
                } else {
                    sJson["hasOutfit"] = false;
                    sJson["name"] = "";
                    sJson["items"] = json::array();
                    sJson["contextTag"] = "";
                    sJson["preferenceStars"] = 3;
                    sJson["controlState"] = 0;
                    sJson["usageCount"] = 0;
                    sJson["lastWornTimestamp"] = 0;
                }
                slotsJson.push_back(sJson);
            }
            wJson["slots"] = slotsJson;
            wheelsJson.push_back(wJson);
        }
        root["wheels"] = wheelsJson;

        std::filesystem::create_directories("Data/SKSE/Plugins");
        std::string tmpPath = g_savePath + ".tmp";
        {
            std::ofstream file(tmpPath);
            if (!file.is_open()) {
                SKSE::log::error("Failed to open temporary save file at {}", tmpPath);
                return;
            }
            file << root.dump(4);
        }
        std::filesystem::rename(tmpPath, g_savePath);
        SKSE::log::info("Data successfully saved to {}", g_savePath);

        SyncWheelOutfitsToAdaptiveWardrobe();
    } catch (const std::exception& e) {
        SKSE::log::error("Exception in SaveDataInternal: {}", e.what());
    }
}

void LoadData() {
    std::lock_guard<std::mutex> lock(g_dataMutex);
    if (!std::filesystem::exists(g_savePath)) {
        SKSE::log::info("No save file found at {}", g_savePath);
        return;
    }
    std::ifstream file(g_savePath);
    if (!file.is_open()) {
        SKSE::log::error("Failed to open save file for reading at {}", g_savePath);
        return;
    }

    try {
        json root = json::parse(file);
        if (root.contains("activeWheel") && root["activeWheel"].is_number()) {
            g_manager.activeWheelIndex = root["activeWheel"].get<size_t>();
        }
        if (root.contains("wheels") && root["wheels"].is_array()) {
            g_manager.wheels.clear();
            for (const auto& wJson : root["wheels"]) {
                Wheel w;
                w.name = wJson.value("name", "Wheel");
                if (wJson.contains("slots") && wJson["slots"].is_array()) {
                    for (const auto& sJson : wJson["slots"]) {
                        Slot s;
                        bool hasOutfit = sJson.value("hasOutfit", false);
                        bool hasItems = sJson.contains("items") && sJson["items"].is_array() && !sJson["items"].empty();

                        if (hasOutfit || hasItems) {
                            Outfit o;
                            o.name = sJson.value("name", "Outfit");
                            o.contextTag = sJson.value("contextTag", "");
                            o.preferenceStars = sJson.value("preferenceStars", static_cast<uint8_t>(3));
                            o.controlState = static_cast<ControlState>(sJson.value("controlState", static_cast<uint8_t>(ControlState::Normal)));
                            o.usageCount = sJson.value("usageCount", 0u);
                            o.lastWornTimestamp = sJson.value("lastWornTimestamp", 0u);

                            if (hasItems) {
                                for (const auto& jItem : sJson["items"]) {
                                    RE::FormID resolvedID = DeserializeFormID(jItem);
                                    if (resolvedID != 0) {
                                        o.itemIDs.push_back(resolvedID);
                                    }
                                }
                            }
                            s.outfit = o;
                        }
                        w.slots.push_back(s);
                    }
                }
                g_manager.wheels.push_back(w);
            }
        }
        if (g_manager.wheels.empty()) {
            Wheel defaultWheel;
            defaultWheel.name = "Wheel 1";
            g_manager.wheels.push_back(defaultWheel);
        }
        if (g_manager.activeWheelIndex >= g_manager.wheels.size()) {
            g_manager.activeWheelIndex = 0;
        }
        SKSE::log::info("Successfully loaded data from disk");

        SyncWheelOutfitsToAdaptiveWardrobe();
    } catch (const std::exception& e) {
        SKSE::log::error("Exception parsing JSON data: {}", e.what());
    }
}

std::string GetWheelJsonInternal(bool isEditMode) {
    if (g_manager.wheels.empty()) {
        Wheel defaultWheel;
        defaultWheel.name = "Wheel 1";
        g_manager.wheels.push_back(defaultWheel);
        g_manager.activeWheelIndex = 0;
    }
    if (g_manager.activeWheelIndex >= g_manager.wheels.size()) {
        g_manager.activeWheelIndex = 0;
    }

    std::set<RE::FormID> currentlyWornSet;
    auto equippedEntries = GetEquippedInventoryEntries();
    for (const auto& entry : equippedEntries) {
        if (entry.object) {
            currentlyWornSet.insert(entry.object->GetFormID());
        }
    }

    const auto& activeWheel = g_manager.wheels[g_manager.activeWheelIndex];
    size_t bestMatchingSlotIndex = SIZE_MAX;
    float bestMatchRatio = 0.0f;
    size_t maxMatches = 0;

    size_t idx = 0;
    for (const auto& slot : activeWheel.slots) {
        if (slot.outfit.has_value() && !slot.outfit->itemIDs.empty()) {
            size_t matches = 0;
            for (RE::FormID fID : slot.outfit->itemIDs) {
                if (currentlyWornSet.find(fID) != currentlyWornSet.end()) {
                    matches++;
                }
            }
            if (matches > 0) {
                float ratio = static_cast<float>(matches) / static_cast<float>(slot.outfit->itemIDs.size());
                if (ratio > bestMatchRatio || (ratio == bestMatchRatio && matches > maxMatches)) {
                    bestMatchRatio = ratio;
                    maxMatches = matches;
                    bestMatchingSlotIndex = idx;
                }
            }
        }
        idx++;
    }

    json data;
    data["editMode"] = isEditMode;
    data["wheelIndex"] = g_manager.activeWheelIndex;
    data["wheelName"] = activeWheel.name;
    data["totalWheels"] = g_manager.wheels.size();
    data["wheelScale"] = g_wheelScale;
    data["wheelOffsetX"] = g_wheelOffsetX;
    data["editKey"] = g_editKey;
    data["editModifierKey"] = g_editModifierKey;
    data["playKey"] = g_playKey;
    data["playModifierKey"] = g_playModifierKey;

    data["autoSystemEnabled"] = g_autoSystemEnabled;
    data["autoCityEnabled"] = g_autoCityEnabled;
    data["autoHomeEnabled"] = g_autoHomeEnabled;
    data["autoCombatEnabled"] = g_autoCombatEnabled;
    data["autoSwimEnabled"] = g_autoSwimEnabled;

    json slotsJson = json::array();
    size_t slotIdx = 0;
    for (const auto& slot : activeWheel.slots) {
        json sJson;
        sJson["index"] = slotIdx;
        sJson["hasOutfit"] = slot.outfit.has_value();
        sJson["name"] = slot.outfit.has_value() ? slot.outfit->name : "Empty";
        sJson["count"] = slot.outfit.has_value() ? slot.outfit->itemIDs.size() : 0;
        sJson["contextTag"] = slot.outfit.has_value() ? slot.outfit->contextTag : "";
        sJson["preferenceStars"] = slot.outfit.has_value() ? slot.outfit->preferenceStars : 3;
        sJson["controlState"] = slot.outfit.has_value() ? static_cast<uint8_t>(slot.outfit->controlState) : 0;

        bool isWorn = (slotIdx == bestMatchingSlotIndex && bestMatchRatio >= 0.5f);
        sJson["isWorn"] = isWorn;
        sJson["isEquipped"] = isWorn;

        slotsJson.push_back(sJson);
        slotIdx++;
    }
    data["slots"] = slotsJson;
    return data.dump();
}

enum class ContextFlags : uint32_t {
    None        = 0,
    CityTown    = 1 << 0,
    PlayerHome  = 1 << 1,
    Combat      = 1 << 2,
    Swim        = 1 << 3,
    Inn         = 1 << 4
};

inline ContextFlags operator|(ContextFlags a, ContextFlags b) {
    return static_cast<ContextFlags>(static_cast<uint32_t>(a) | static_cast<uint32_t>(b));
}

inline ContextFlags operator&(ContextFlags a, ContextFlags b) {
    return static_cast<ContextFlags>(static_cast<uint32_t>(a) & static_cast<uint32_t>(b));
}

inline ContextFlags& operator|=(ContextFlags& a, ContextFlags b) {
    a = a | b;
    return a;
}

inline bool HasContext(ContextFlags mask, ContextFlags flag) {
    return (static_cast<uint32_t>(mask) & static_cast<uint32_t>(flag)) != 0;
}

enum class ControlState : uint8_t {
    Normal = 0,
    Favorite,
    Forced,
    Excluded,
    Locked
};

struct SuitabilityRule {
    ContextFlags contextMask = ContextFlags::None;
    uint8_t score = 1;
};

NLOHMANN_DEFINE_TYPE_NON_INTRUSIVE(SuitabilityRule, contextMask, score)

struct DynamicVariables {
    uint32_t usageCount = 0;
    float freshnessMultiplier = 1.5f;
    uint32_t lastWornTimestamp = 0;
};

NLOHMANN_DEFINE_TYPE_NON_INTRUSIVE(DynamicVariables, usageCount, freshnessMultiplier, lastWornTimestamp)

struct AdaptiveOutfit {
    std::string id;
    std::string name;
    std::vector<RE::FormID> itemIDs;
    std::vector<SuitabilityRule> suitabilities;
    uint8_t preferenceStars = 3;
    ControlState controlState = ControlState::Normal;
    DynamicVariables dynamicVars;
};

inline void to_json(json& j, const AdaptiveOutfit& o) {
    j = json{
        {"id", o.id},
        {"name", o.name},
        {"itemIDs", o.itemIDs},
        {"suitabilities", o.suitabilities},
        {"preferenceStars", o.preferenceStars},
        {"controlState", static_cast<uint8_t>(o.controlState)},
        {"dynamicVars", o.dynamicVars}
    };
}

inline void from_json(const json& j, AdaptiveOutfit& o) {
    j.at("id").get_to(o.id);
    j.at("name").get_to(o.name);
    j.at("itemIDs").get_to(o.itemIDs);
    j.at("suitabilities").get_to(o.suitabilities);
    j.at("preferenceStars").get_to(o.preferenceStars);
    uint8_t cs = j.at("controlState").get<uint8_t>();
    o.controlState = static_cast<ControlState>(cs);
    j.at("dynamicVars").get_to(o.dynamicVars);
}

    enum class ConsoleMessageLevel : uint8_t
    {
        Log = 0,
        Warning,
        Error,
        Debug,
        Info
    };

    // Console message callback.
    typedef void (*ConsoleMessageCallback)(PrismaView view, ConsoleMessageLevel level, const char *message);

    // PrismaUI modder interface v1
    class IVPrismaUI1
    {
    protected:
        ~IVPrismaUI1() = default;

    public:
        // Create view.
        virtual PrismaView CreateView(const char *htmlPath,
                                      OnDomReadyCallback onDomReadyCallback = nullptr) noexcept = 0;

        // Send JS code to UI.
        virtual void Invoke(PrismaView view, const char *script, JSCallback callback = nullptr) noexcept = 0;

        // Call JS function through JS Interop API (best performance).
        virtual void InteropCall(PrismaView view, const char *functionName, const char *argument) noexcept = 0;

        // Register JS listener.
        virtual void RegisterJSListener(PrismaView view, const char *functionName,
                                        JSListenerCallback callback) noexcept = 0;

        // Returns true if view has focus.
        virtual bool HasFocus(PrismaView view) noexcept = 0;

        // Set focus on view.
        virtual bool Focus(PrismaView view, bool pauseGame = false, bool disableFocusMenu = false) noexcept = 0;

        // Remove focus from view.
        virtual void Unfocus(PrismaView view) noexcept = 0;

        // Show a hidden view.
        virtual void Show(PrismaView view) noexcept = 0;

        // Hide a visible view.
        virtual void Hide(PrismaView view) noexcept = 0;

        // Returns true if view is hidden.
        virtual bool IsHidden(PrismaView view) noexcept = 0;

        // Get scroll size in pixels.
        virtual int GetScrollingPixelSize(PrismaView view) noexcept = 0;

        // Set scroll size in pixels.
        virtual void SetScrollingPixelSize(PrismaView view, int pixelSize) noexcept = 0;

        // Returns true if view exists.
        virtual bool IsValid(PrismaView view) noexcept = 0;

        // Completely destroy view.
        virtual void Destroy(PrismaView view) noexcept = 0;

        // Set view order.
        virtual void SetOrder(PrismaView view, int order) noexcept = 0;

        // Get view order.
        virtual int GetOrder(PrismaView view) noexcept = 0;

        // Create inspector view for debugging.
        virtual void CreateInspectorView(PrismaView view) noexcept = 0;

        // Show or hide the inspector overlay.
        virtual void SetInspectorVisibility(PrismaView view, bool visible) noexcept = 0;

        // Returns true if inspector is visible.
        virtual bool IsInspectorVisible(PrismaView view) noexcept = 0;

        // Set inspector window position and size.
        virtual void SetInspectorBounds(PrismaView view, float topLeftX, float topLeftY, unsigned int width,
                                        unsigned int height) noexcept = 0;

        // Returns true if any view has active focus.
        virtual bool HasAnyActiveFocus() noexcept = 0;
    };

    // PrismaUI modder interface v2 (extends v1)
    class IVPrismaUI2 : public IVPrismaUI1
    {

        ~IVPrismaUI2() = default;

        // Pass nullptr to unregister.
        virtual void RegisterConsoleCallback(PrismaView view, ConsoleMessageCallback callback) noexcept = 0;
    };

    // Maps interface types to InterfaceVersion enum values.
    // compile-time constraint -- only request interface versions that actually exist.
    template <typename T>
    struct InterfaceVersionMap;

    template <>
    struct InterfaceVersionMap<IVPrismaUI1>
    {
        static constexpr InterfaceVersion version = InterfaceVersion::V1;
    };

    template <>
    struct InterfaceVersionMap<IVPrismaUI2>
    {
        static constexpr InterfaceVersion version = InterfaceVersion::V2;
    };

    typedef void *(*RequestPluginAPIFunc)(InterfaceVersion interfaceVersion);

    /// Request the PrismaUI API interface.
    /// Recommended: Send your request during or after SKSEMessagingInterface::kMessage_PostLoad to make sure the dll
    /// has already been loaded
    [[nodiscard]] inline void *RequestPluginAPI(InterfaceVersion a_interfaceVersion = InterfaceVersion::V1)
    {
        auto pluginHandle = GetModuleHandle("PrismaUI.dll");
        if (!pluginHandle)
        {
            return nullptr;
        }

        auto requestAPIFunction =
            reinterpret_cast<RequestPluginAPIFunc>(GetProcAddress(pluginHandle, "RequestPluginAPI"));

        if (requestAPIFunction)
        {
            return requestAPIFunction(a_interfaceVersion);
        }

        return nullptr;
    }

    /// Request a specific PrismaUI API interface version.
    /// Returns nullptr if the loaded PrismaUI DLL does not support the requested version.
    ///
    /// Usage:
    ///   auto* m_prismaUI   = PRISMA_UI_API::RequestPluginAPI<PRISMA_UI_API::IVPrismaUI1>();
    ///   auto* m_prismaUIv2 = PRISMA_UI_API::RequestPluginAPI<PRISMA_UI_API::IVPrismaUI2>();
    template <typename T>
    [[nodiscard]] inline T *RequestPluginAPI()
    {
        return static_cast<T *>(RequestPluginAPI(InterfaceVersionMap<T>::version));
    }

namespace Wheeler {

    inline RE::InventoryEntryData* GetSelectedInventoryItem() {
        auto ui = RE::UI::GetSingleton();
        if (!ui) return nullptr;

        auto invMenu = ui->GetMenu<RE::InventoryMenu>(RE::InventoryMenu::MENU_NAME);

        if (invMenu && invMenu->uiMovie) {
            RE::GFxValue indexVal;

            bool success = invMenu->uiMovie->GetVariable(&indexVal, "_root.Menu_mc.inventoryLists.itemList.iSelectedIndex");

            if (success && indexVal.IsNumber()) {
                std::size_t idx = static_cast<std::size_t>(indexVal.GetNumber());

                auto itemList = invMenu->GetRuntimeData().itemList;

                if (itemList && idx < itemList->items.size()) {
                    auto item = itemList->items[idx];
                    if (item) {
                        return item->data.objDesc;
                    }
                }
            }
        }
        return nullptr;
    }
